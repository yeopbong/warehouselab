cmake_minimum_required(VERSION 3.16)
project(tmo_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TMO_SOURCE_DIR)
  message(FATAL_ERROR "Set TMO_SOURCE_DIR to the pinned WPPL checkout")
endif()
find_package(Boost REQUIRED COMPONENTS program_options filesystem log log_setup iostreams)
find_package(spdlog REQUIRED)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
file(GLOB CORE_SOURCES "${TMO_SOURCE_DIR}/src/*.cpp")
list(FILTER CORE_SOURCES EXCLUDE REGEX "/(driver|py_driver|py_sim|analyze_results)\\.cpp$")
file(GLOB_RECURSE PLANNER_SOURCES "${TMO_SOURCE_DIR}/src/LaCAM2/*.cpp" "${TMO_SOURCE_DIR}/src/LNS/*.cpp" "${TMO_SOURCE_DIR}/src/util/*.cpp")
add_executable(tmo-run bridge.cpp ${CORE_SOURCES} ${PLANNER_SOURCES})
target_include_directories(tmo-run PRIVATE "${TMO_SOURCE_DIR}/inc" ${Boost_INCLUDE_DIRS})
target_compile_definitions(tmo-run PRIVATE NO_ROT MAP_OPT PYTHON=0 BOOST_LOG_DYN_LINK)
target_link_libraries(tmo-run PRIVATE ${Boost_LIBRARIES} spdlog::spdlog OpenMP::OpenMP_CXX Threads::Threads)
